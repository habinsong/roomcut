/*
 * test_heartbeat.cpp — verifies the HEALTH_CHECK liveness path
 * (DEVELOPMENT_PLAN.md §4.1 heartbeat, §4.2 timeout → peer lost).
 *
 *   1. Live peer: a responder thread echoes one probe with its state; the
 *      prober gets KERN_SUCCESS and reads that state back.
 *   2. Dead peer: probing a health port with no responder times out
 *      (MACH_RCV_TIMED_OUT) — this is exactly the signal the driver uses to
 *      transition Streaming → EngineLost.
 */
#include "Heartbeat.hpp"
#include "RoomcutTransport.h"

#include <cstdio>
#include <cstring>
#include <pthread.h>
#include <cstddef>
#include <atomic>
#include <thread>

#include <mach/mach.h>

static std::atomic<int> g_failures{0};
#define CHECK(cond, msg) do { \
    if (!(cond)) { fprintf(stderr, "FAIL: %s (%s:%d)\n", (msg), __FILE__, __LINE__); g_failures++; } \
} while (0)

using namespace roomcut;

struct RespCtx {
    mach_port_t healthPort;
    uint32_t    state;
    bool        ready;
    pthread_mutex_t mtx;
    pthread_cond_t  cv;
};

// Responder: receive one probe, reply with its state, exit.
static void* responderThread(void* arg) {
    RespCtx* c = (RespCtx*)arg;
    RoomcutHealthRequest req;
    kern_return_t kr = heartbeatReceive(c->healthPort, 2000, &req);
    CHECK(kr == KERN_SUCCESS, "responder receives probe");
    if (kr == KERN_SUCCESS) {
        kr = heartbeatRespond(req, c->state);
        CHECK(kr == KERN_SUCCESS, "responder sends reply");
    }
    return nullptr;
}

static void test_live_peer() {
    mach_port_t self = mach_task_self();
    mach_port_t health = MACH_PORT_NULL;
    CHECK(mach_port_allocate(self, MACH_PORT_RIGHT_RECEIVE, &health) == KERN_SUCCESS, "alloc health port");
    CHECK(mach_port_insert_right(self, health, health, MACH_MSG_TYPE_MAKE_SEND) == KERN_SUCCESS, "insert send right");

    RespCtx c;
    std::memset(&c, 0, sizeof(c));
    c.healthPort = health;
    c.state = ROOMCUT_STATE_RUNNING;

    pthread_t t;
    pthread_create(&t, nullptr, responderThread, &c);

    uint32_t peerState = 0xFFFFFFFF;
    kern_return_t kr = heartbeatProbe(health, 42, 2000, &peerState);
    CHECK(kr == KERN_SUCCESS, "live probe succeeds");
    CHECK(peerState == ROOMCUT_STATE_RUNNING, "probe reads peer state");

    pthread_join(t, nullptr);
    mach_port_mod_refs(self, health, MACH_PORT_RIGHT_RECEIVE, -1);
}

static void test_dead_peer() {
    mach_port_t self = mach_task_self();
    mach_port_t health = MACH_PORT_NULL;
    CHECK(mach_port_allocate(self, MACH_PORT_RIGHT_RECEIVE, &health) == KERN_SUCCESS, "alloc dead health port");
    CHECK(mach_port_insert_right(self, health, health, MACH_MSG_TYPE_MAKE_SEND) == KERN_SUCCESS, "insert send right (dead)");

    // No responder: the probe sends fine but no reply arrives → timeout.
    uint32_t peerState = 0;
    kern_return_t kr = heartbeatProbe(health, 7, 200, &peerState);
    CHECK(kr == MACH_RCV_TIMED_OUT, "dead probe times out (peer lost signal)");

    mach_port_mod_refs(self, health, MACH_PORT_RIGHT_RECEIVE, -1);
}

enum class ReplyCase { current, legacy, truncated, wrongID, wrongType, wrongSequence,
                       invalidState, unexpectedRight, complex };

static void test_reply_boundaries(bool driver) {
    for (const auto kind : {ReplyCase::current, ReplyCase::legacy, ReplyCase::truncated,
                           ReplyCase::wrongID, ReplyCase::wrongType, ReplyCase::wrongSequence,
                           ReplyCase::invalidState, ReplyCase::unexpectedRight, ReplyCase::complex}) {
        mach_port_t service = MACH_PORT_NULL, extra = MACH_PORT_NULL;
        CHECK(mach_port_allocate(mach_task_self(), MACH_PORT_RIGHT_RECEIVE, &service) == KERN_SUCCESS, "allocate boundary service");
        CHECK(mach_port_insert_right(mach_task_self(), service, service, MACH_MSG_TYPE_MAKE_SEND) == KERN_SUCCESS, "create boundary service right");
        CHECK(mach_port_allocate(mach_task_self(), MACH_PORT_RIGHT_RECEIVE, &extra) == KERN_SUCCESS, "allocate carried right");
        std::thread server([&] {
            RoomcutHealthMsgBuffer received{};
            const auto kr = mach_msg(&received.request.header, MACH_RCV_MSG | MACH_RCV_TIMEOUT,
                0, sizeof(received), service, 2000, MACH_PORT_NULL);
            CHECK(kr == KERN_SUCCESS, "receive boundary probe");
            if (kr != KERN_SUCCESS) return;
            RoomcutHealthReply reply{};
            reply.header.msgh_bits = MACH_MSGH_BITS(MACH_MSG_TYPE_MOVE_SEND_ONCE, 0);
            reply.header.msgh_remote_port = received.request.header.msgh_remote_port;
            reply.header.msgh_id = ROOMCUT_MSG_HEALTH_CHECK;
            reply.msgType = ROOMCUT_MSG_HEALTH_CHECK;
            reply.sequence = received.request.sequence;
            reply.state = ROOMCUT_STATE_RUNNING;
            mach_msg_size_t size = sizeof(reply);
            if (kind == ReplyCase::legacy) size = offsetof(RoomcutHealthReply, availableRateCount);
            if (kind == ReplyCase::truncated) size = offsetof(RoomcutHealthReply, state);
            if (kind == ReplyCase::wrongID) reply.header.msgh_id = ROOMCUT_MSG_HELLO;
            if (kind == ReplyCase::wrongType) reply.msgType = ROOMCUT_MSG_HELLO;
            if (kind == ReplyCase::wrongSequence) ++reply.sequence;
            if (kind == ReplyCase::invalidState) reply.state = 0xFFFFFFFF;
            if (kind == ReplyCase::unexpectedRight) {
                reply.header.msgh_bits = MACH_MSGH_BITS(MACH_MSG_TYPE_MOVE_SEND_ONCE, MACH_MSG_TYPE_MAKE_SEND);
                reply.header.msgh_local_port = extra;
            }
            if (kind == ReplyCase::complex) {
                struct {
                    mach_msg_header_t header;
                    mach_msg_body_t body;
                    mach_msg_port_descriptor_t descriptor;
                    uint32_t msgType, sequence, state;
                } message{};
                message.header = reply.header;
                message.header.msgh_bits |= MACH_MSGH_BITS_COMPLEX;
                message.body.msgh_descriptor_count = 1;
                message.descriptor.name = extra;
                message.descriptor.disposition = MACH_MSG_TYPE_MAKE_SEND;
                message.descriptor.type = MACH_MSG_PORT_DESCRIPTOR;
                message.msgType = reply.msgType;
                message.sequence = reply.sequence;
                message.state = reply.state;
                CHECK(mach_msg(&message.header, MACH_SEND_MSG | MACH_SEND_TIMEOUT,
                    sizeof(message), 0, MACH_PORT_NULL, 2000, MACH_PORT_NULL) == KERN_SUCCESS, "send descriptor-bearing reply");
            } else {
                CHECK(mach_msg(&reply.header, MACH_SEND_MSG | MACH_SEND_TIMEOUT,
                    size, 0, MACH_PORT_NULL, 2000, MACH_PORT_NULL) == KERN_SUCCESS, "send boundary reply");
            }
        });
        uint32_t state = 0xABCDEF;
        const auto result = driver ? Roomcut_TransportProbePort(service, 42, 2000)
                                   : heartbeatProbe(service, 42, 2000, &state);
        server.join();
        const bool valid = kind == ReplyCase::current || kind == ReplyCase::legacy;
        CHECK(result == (valid ? KERN_SUCCESS : KERN_FAILURE), "only complete matching simple heartbeat replies succeed");
        CHECK(state == (valid && !driver ? ROOMCUT_STATE_RUNNING : 0xABCDEF), "rejected replies preserve the caller's state");
        mach_port_urefs_t refs = 0;
        mach_port_get_refs(mach_task_self(), extra, MACH_PORT_RIGHT_SEND, &refs);
        CHECK(refs == 0, "unexpected reply rights are released");
        if (refs) mach_port_mod_refs(mach_task_self(), extra, MACH_PORT_RIGHT_SEND, -refs);
        mach_port_mod_refs(mach_task_self(), extra, MACH_PORT_RIGHT_RECEIVE, -1);
        mach_port_deallocate(mach_task_self(), service);
        mach_port_mod_refs(mach_task_self(), service, MACH_PORT_RIGHT_RECEIVE, -1);
    }
}

static void test_request_boundaries() {
    for (int variant = 0; variant < 5; ++variant) {
        mach_port_t service = MACH_PORT_NULL, replyPort = MACH_PORT_NULL;
        mach_port_allocate(mach_task_self(), MACH_PORT_RIGHT_RECEIVE, &service);
        mach_port_insert_right(mach_task_self(), service, service, MACH_MSG_TYPE_MAKE_SEND);
        mach_port_allocate(mach_task_self(), MACH_PORT_RIGHT_RECEIVE, &replyPort);
        RoomcutHealthRequest request{};
        request.header.msgh_bits = MACH_MSGH_BITS(MACH_MSG_TYPE_COPY_SEND, MACH_MSG_TYPE_MAKE_SEND_ONCE);
        request.header.msgh_remote_port = service;
        request.header.msgh_local_port = replyPort;
        request.header.msgh_id = ROOMCUT_MSG_HEALTH_CHECK;
        request.msgType = ROOMCUT_MSG_HEALTH_CHECK;
        request.sequence = 42;
        mach_msg_size_t size = sizeof(request);
        if (variant == 1) size = offsetof(RoomcutHealthRequest, sequence);
        if (variant == 2) request.header.msgh_id = ROOMCUT_MSG_HELLO;
        if (variant == 3) request.msgType = ROOMCUT_MSG_HELLO;
        if (variant == 4) {
            request.header.msgh_bits = MACH_MSGH_BITS(MACH_MSG_TYPE_COPY_SEND, MACH_MSG_TYPE_MAKE_SEND);
        }
        CHECK(mach_msg(&request.header, MACH_SEND_MSG | MACH_SEND_TIMEOUT, size,
            0, MACH_PORT_NULL, 2000, MACH_PORT_NULL) == KERN_SUCCESS, "send request boundary case");
        RoomcutHealthRequest received{};
        received.sequence = 0xABCDEF;
        const auto result = heartbeatReceive(service, 2000, &received);
        CHECK(result == (variant == 0 ? KERN_SUCCESS : KERN_FAILURE), "only complete matching send-once requests succeed");
        CHECK(received.sequence == (variant == 0 ? 42 : 0xABCDEF), "invalid request does not overwrite caller data");
        if (result == KERN_SUCCESS) mach_msg_destroy(&received.header);
        mach_port_urefs_t refs = 0;
        mach_port_get_refs(mach_task_self(), replyPort, MACH_PORT_RIGHT_SEND, &refs);
        CHECK(refs == 0, "rejected requests release carried send rights");
        if (refs) mach_port_mod_refs(mach_task_self(), replyPort, MACH_PORT_RIGHT_SEND, -refs);
        mach_port_mod_refs(mach_task_self(), replyPort, MACH_PORT_RIGHT_RECEIVE, -1);
        mach_port_deallocate(mach_task_self(), service);
        mach_port_mod_refs(mach_task_self(), service, MACH_PORT_RIGHT_RECEIVE, -1);
    }
}

int main() {
    test_live_peer();
    test_dead_peer();
    test_reply_boundaries(false);
    test_reply_boundaries(true);
    test_request_boundaries();

    if (g_failures == 0) { printf("all heartbeat tests passed\n"); return 0; }
    fprintf(stderr, "%d heartbeat check(s) failed\n", g_failures.load());
    return 1;
}
