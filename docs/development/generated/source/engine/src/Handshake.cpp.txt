bool engineNegotiateHello(const RoomcutHelloRequest& request, RoomcutFormatNegotiation& granted) {
    if (!roomcut_hello_request_valid(&request)) return false;
    RoomcutFormatNegotiation result{};
    result.sampleRate = roomcut_sr_supported(request.requested.sampleRate)
        ? request.requested.sampleRate : ROOMCUT_SR_48000;
    result.channels = ROOMCUT_MVP_CHANNELS;
    result.channelLayout = ROOMCUT_LAYOUT_STEREO;
    result.internalFormat = ROOMCUT_INTERNAL_FORMAT_F32;
    result.bufferFrameSize = request.requested.bufferFrameSize;
    result.latencyClass = request.requested.latencyClass;
    auto capacity = request.requested.capacityFrames;
    if (!roomcut_is_pow2(capacity) || capacity < 256 || capacity > 65536)
        capacity = ROOMCUT_DEFAULT_CAPACITY_FRAMES;
    // Keep the existing ~85 ms jitter cushion at high sample rates.
    while (capacity < result.sampleRate / 12u && capacity < 65536u) capacity <<= 1;
    result.capacityFrames = capacity;
    granted = result;
    return true;
}

kern_return_t engineReplyHello(const RoomcutHelloRequest& request,
                               const RingRegion& region,
                               const RoomcutFormatNegotiation& granted,
                               const uint32_t* availableRates,
                               uint32_t availableRateCount) {
    if (!roomcut_hello_request_valid(&request) || !roomcut_hello_format_valid(&granted) ||
        !region.valid() || region.memoryEntry() == MACH_PORT_NULL ||
        !roomcut_ring_validate(region.header()) ||
        region.header()->capacityFrames != granted.capacityFrames ||
        region.header()->channels != granted.channels || region.header()->sampleRate != granted.sampleRate) {
        return KERN_INVALID_ARGUMENT;
    }
    mach_port_t replyPort = request.header.msgh_remote_port;
    if (replyPort == MACH_PORT_NULL) {
        return KERN_INVALID_ARGUMENT;
    }

    RoomcutHelloReply reply;
    std::memset(&reply, 0, sizeof(reply));

    // Complex message: header bits MACH_MSGH_BITS_COMPLEX, one descriptor.
    reply.header.msgh_bits        = MACH_MSGH_BITS(MACH_MSG_TYPE_MOVE_SEND_ONCE, 0)
                                  | MACH_MSGH_BITS_COMPLEX;
    reply.header.msgh_size        = sizeof(reply);
    reply.header.msgh_remote_port = replyPort;          // send the reply here
    reply.header.msgh_local_port  = MACH_PORT_NULL;
    reply.header.msgh_id          = ROOMCUT_MSG_HELLO;

    reply.body.msgh_descriptor_count = 1;
    reply.memoryEntry.name        = region.memoryEntry();
    reply.memoryEntry.disposition = MACH_MSG_TYPE_COPY_SEND; // mint a ref for receiver
    reply.memoryEntry.type        = MACH_MSG_PORT_DESCRIPTOR;

    reply.msgType = ROOMCUT_MSG_HELLO;
    reply.status  = 0;
    reply.granted = granted;

    reply.availableRateCount = 0;
    if (availableRates != nullptr && availableRateCount > 0) {
        uint32_t n = availableRateCount > ROOMCUT_MAX_RATES
                         ? (uint32_t)ROOMCUT_MAX_RATES : availableRateCount;
        for (uint32_t i = 0; i < n; ++i) reply.availableRates[i] = availableRates[i];
        reply.availableRateCount = n;
    }

    const auto result = mach_msg(&reply.header, MACH_SEND_MSG, sizeof(reply), 0,
                                 MACH_PORT_NULL, MACH_MSG_TIMEOUT_NONE, MACH_PORT_NULL);
    roomcut_destroy_unsent_message(&reply.header, result);
    return result;
}

kern_return_t driverSendHelloAndReceive(mach_port_t servicePort,
                                        const RoomcutFormatNegotiation& requested,
                                        RingRegion& outRegion,
                                        RoomcutFormatNegotiation* outGranted,
                                        uint32_t* outRates,
                                        uint32_t* outRateCount,
                                        uint32_t timeoutMs) {
    mach_port_t self = mach_task_self();

    // Make a reply port (receive right) the engine will send its reply to.
    mach_port_t replyPort = MACH_PORT_NULL;
    kern_return_t kr = mach_port_allocate(self, MACH_PORT_RIGHT_RECEIVE, &replyPort);
    if (kr != KERN_SUCCESS) {
        return kr;
    }

    // Send the HELLO request, carrying a send-once right to our reply port.
    RoomcutHelloRequest req;
    std::memset(&req, 0, sizeof(req));
    req.header.msgh_bits        = MACH_MSGH_BITS(MACH_MSG_TYPE_COPY_SEND,
                                                 MACH_MSG_TYPE_MAKE_SEND_ONCE);
    req.header.msgh_size        = sizeof(req);
    req.header.msgh_remote_port = servicePort;   // destination: engine service
    req.header.msgh_local_port  = replyPort;     // engine replies here
    req.header.msgh_id          = ROOMCUT_MSG_HELLO;
    req.msgType                 = ROOMCUT_MSG_HELLO;
    req.protocolVersion         = ROOMCUT_IPC_VERSION;
    req.requested               = requested;

    kr = mach_msg(&req.header, MACH_SEND_MSG | MACH_SEND_TIMEOUT, sizeof(req), 0,
                  MACH_PORT_NULL, timeoutMs, MACH_PORT_NULL);
    if (kr != KERN_SUCCESS) {
        roomcut_destroy_unsent_message(&req.header, kr);
        mach_port_mod_refs(self, replyPort, MACH_PORT_RIGHT_RECEIVE, -1);
        return kr;
    }

    // Receive the reply (with the memory-entry port descriptor).
    RoomcutHelloMsgBuffer buf;
    std::memset(&buf, 0, sizeof(buf));
    kr = mach_msg(&buf.reply.header, MACH_RCV_MSG | MACH_RCV_TIMEOUT, 0, sizeof(buf),
                  replyPort, timeoutMs, MACH_PORT_NULL);
    if (kr != KERN_SUCCESS) {
        mach_port_mod_refs(self, replyPort, MACH_PORT_RIGHT_RECEIVE, -1);
        return kr;
    }

    if (!roomcut_hello_reply_valid(&buf.reply)) {
        mach_msg_destroy(&buf.reply.header);
        mach_port_mod_refs(self, replyPort, MACH_PORT_RIGHT_RECEIVE, -1);
        return KERN_FAILURE;
    }

    // Map exactly the negotiated extent and verify its header before publishing
    // any output. mapFromPort owns the memory-entry right only on success.
    RingRegion mapped;
    kr = mapped.mapFromPort(buf.reply.memoryEntry.name, buf.reply.granted);
    if (kr != KERN_SUCCESS) {
        mach_msg_destroy(&buf.reply.header);
        mach_port_mod_refs(self, replyPort, MACH_PORT_RIGHT_RECEIVE, -1);
        return kr;
    }
    outRegion = static_cast<RingRegion&&>(mapped);
    if (outGranted) {
        *outGranted = buf.reply.granted;
    }
    if (outRateCount) {
        uint32_t n = buf.reply.header.msgh_size >= sizeof(RoomcutHelloReply)
            ? buf.reply.availableRateCount : 0;
        if (n > ROOMCUT_MAX_RATES) n = ROOMCUT_MAX_RATES;
        if (outRates) {
            for (uint32_t i = 0; i < n; ++i) outRates[i] = buf.reply.availableRates[i];
        }
        *outRateCount = n;
    }

    mach_port_mod_refs(self, replyPort, MACH_PORT_RIGHT_RECEIVE, -1);
    return kr;
}

} // namespace roomcut
