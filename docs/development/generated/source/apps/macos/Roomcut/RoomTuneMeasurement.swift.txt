import RoomcutCore

// App composition: the observable workflow owns sequencing; the session owns media IO.
typealias RoomTuneMeasurement = RoomTuneWorkflow

extension RoomTuneWorkflow {
    convenience init() {
        let audio: RoomTuneAudio = AppLaunch.fixtureKind == .uiRoomTune
            ? FixtureRoomTuneAudio() : RoomTuneAudioSession()
        self.init(audio: audio)
    }
}
